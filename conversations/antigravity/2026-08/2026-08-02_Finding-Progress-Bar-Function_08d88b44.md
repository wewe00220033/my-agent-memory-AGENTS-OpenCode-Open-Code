import os
import sys
import time

# تفعيل ألوان ANSI في كونسول ويندوز
os.system("")

def get_rainbow_color(percent):
    """مصفوفة ألوان ANSI 256 لتوليد لون مختلف لكل 1%"""
    spectrum = [
        51, 50, 49, 48, 47, 46, 45, 44, 43, 42,
        41, 40, 39, 38, 37, 36, 35, 34, 33, 32,
        75, 74, 73, 72, 71, 70, 69, 68, 67, 66,
        99, 98, 97, 96, 95, 94, 93, 92, 91, 90,
        129, 128, 127, 126, 125, 124, 123, 122, 121, 120,
        165, 164, 163, 162, 161, 160, 159, 158, 157, 156,
        201, 200, 199, 198, 197, 196, 195, 194, 193, 192,
        207, 206, 205, 204, 203, 202, 208, 209, 214, 215,
        220, 221, 222, 223, 226, 227, 228, 229, 118, 119,
        82, 83, 84, 85, 86, 87, 14, 15, 11, 9
    ]
    idx = max(0, min(99, int(percent)))
    return f"\033[38;5;{spectrum[idx]}m"

def show_progress_bar(duration_seconds=10):
    """محاكاة شريط التقدم بالثواني"""
    RESET = "\033[0m"
    start_time = time.time()

    while True:
        elapsed = time.time() - start_time
        if elapsed >= duration_seconds:
            break

        remaining = max(0, int(duration_seconds - elapsed))
        pct = min(100.0, (elapsed / duration_seconds) * 100.0)

        color = get_rainbow_color(pct)
        bar_len = 20
        filled = int((pct / 100.0) * bar_len)
        bar = "█" * filled + "░" * (bar_len - filled)

        sys.stdout.write(
            f"\r{color}🤖 [Task Running] ⏳ Processing... "
            f"[{remaining:3d}s remaining] [{bar}] {pct:5.1f}%{RESET}"
        )
        sys.stdout.flush()
        time.sleep(0.1)

    # مسح الشريط بعد الانتهاء
    sys.stdout.write("\r" + " " * 85 + "\r")
    sys.stdout.flush()
    print("✅ تم الانتهاء بنجاح!")

# تشغيل للتجربة (10 ثواني)
if __name__ == "__main__":
    show_progress_bar(10)
