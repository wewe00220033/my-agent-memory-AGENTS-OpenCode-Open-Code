الإجابة الصحيحة فعلاً **302 (Redirect)** مش 200، لأن:

1. **السيرفر بيبعت `Set-Cookie` مع الـ 302** عشان يظبط session cookie جديد
2. **الـ 200 اللي ظهر** معناه إن `requests.Session()` عمل redirect تلقائي وأخد الـ cookie الجديد

**مشكلة محتملة:** ممكن يكون الـ redirect عمل auto-follow ورجع 200، بس **الـ Set-Cookie مش بتظهر في response.headers** لأنها جوا الـ redirect flow.

def build_session():
    session = requests.Session()

    session.headers.update({
        "User-Agent": "Mozilla/5.0 (Linux; Android 10; K) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/147.0.0.0 Mobile Safari/537.36",
        "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
        "Accept-Language": "en-US,en;q=0.9,ar;q=0.8",
        "Referer": TOOLS_URL,
        "Origin": "https://followersize.net",
    })

    initial_cookies = {
        "_gid": "ضع القيمة الحالية",
        "fpestid": "ضع القيمة الحالية",
        "_ga": "ضع القيمة الحالية",
        "_ga_R8Z73JY5ML": "ضع القيمة الحالية",
        "c1279ceb142e4de0024d80a0118219d4": "ضع قيمة session الحالية",
    }

    requests.utils.add_dict_to_cookiejar(session.cookies, initial_cookies)
    return session


def refresh_session(session):
    payload = {
        # نفس بيانات الـ POST الحقيقية من test_session
        # مثال:
        # "username": ".....",
        # "user_id": ".....",
        # "submit": "1",
    }

    response = session.post(
        FIND_USER_URL,
        data=payload,
        allow_redirects=False,
        timeout=15
    )

    return response


def keep_session_alive():
    session = build_session()

    total_runs = (TOTAL_HOURS * 60) // 14

    print("🚀 بدء الحفاظ على الجلسة لمدة 24 ساعة")

    for i in range(1, total_runs + 1):
        print(f"🔄 محاولة {i}/{total_runs}")

        try:
            response = refresh_session(session)

            if response.status_code == 302:
                print(f"✅ تم تجديد الجلسة فعليًا (Status: {response.status_code})")
                location = response.headers.get("Location")
                if location:
                    print(f"↪️ Redirect إلى: {location}")

            elif response.status_code == 200:
                print("⚠️ السيرفر رجّع 200، وده مش دليل كافي على تجديد الجلسة")

            else:
                print(f"❌ رد غير متوقع من السيرفر: {response.status_code}")

            print(f"🍪 الكوكيز الحالية: {session.cookies.get_dict()}")

        except Exception as e:
            print(f"❌ خطأ أثناء تحديث الجلسة: {e}")

        if i < total_runs:
            print("⏳ انتظار 14 دقيقة...\n")
            time.sleep(INTERVAL_SECONDS)

    print("🎉 انتهت دورة الـ 24 ساعة")


if __name__ == "__main__":
