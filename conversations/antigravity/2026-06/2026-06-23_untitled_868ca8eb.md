async function simulateSlider({
    element,
    direction = "right",
    distance = null,
    duration = 800,
    steps = 30
}) {

    if (!element) return false;

    const rect = element.getBoundingClientRect();

    const startX = rect.left + rect.width / 2;
    const startY = rect.top + rect.height / 2;

    const moveDistance =
        distance ??
        Math.max(
            100,
            element.parentElement?.clientWidth || 200
        );

    const endX =
        direction === "left"
            ? startX - moveDistance
            : startX + moveDistance;

    element.dispatchEvent(
        new PointerEvent("pointerdown", {
            bubbles: true,
            clientX: startX,
            clientY: startY
        })
    );

    for (let i = 1; i <= steps; i++) {

        const progress = i / steps;

        const currentX =
            startX +
            (endX - startX) * progress;

        element.dispatchEvent(
            new PointerEvent("pointermove", {
                bubbles: true,
                clientX: currentX,
                clientY: startY
            })
        );

        await new Promise(r =>
            setTimeout(r, duration / steps)
        );
    }

    element.dispatchEvent(
        new PointerEvent("pointerup", {
            bubbles: true,
            clientX: endX,
            clientY: startY
        })
    );

    return true;
}

function createDomSlider({
  track,
  handle,
  fill = null,
  min = 0,
  max = 100,
  initialValue = 0,
  step = 1,
  smooth = true,
  duration = 180,
  onChange = () => {},
  onStart = () => {},
  onEnd = () => {}
}) {
  if (!track || !handle) {
    throw new Error("track و handle مطلوبين");
  }

  let dragging = false;
  let rafId = null;
  let currentPx = 0;
  let targetPx = 0;

  const getMaxPx = () => Math.max(0, track.clientWidth - handle.offsetWidth);

  const clamp = (value, min, max) => Math.min(Math.max(value, min), max);

  const valueToPx = (value) => {
    const ratio = (value - min) / (max - min || 1);
    return clamp(ratio * getMaxPx(), 0, getMaxPx());
  };

  const pxToValue = (px) => {
    const ratio = getMaxPx() === 0 ? 0 : px / getMaxPx();
    const raw = min + ratio * (max - min);
    return Math.round(raw / step) * step;
  };

  const render = (px) => {
    currentPx = clamp(px, 0, getMaxPx());
    handle.style.left = `${currentPx}px`;

    if (fill) {
      fill.style.width = `${currentPx + handle.offsetWidth / 2}px`;
    }

    onChange({
      px: currentPx,
      value: pxToValue(currentPx),
      percent: getMaxPx() === 0 ? 0 : currentPx / getMaxPx()
    });
  };

  const animateTo = (newPx) => {
    targetPx = clamp(newPx, 0, getMaxPx());

    if (!smooth) {
      render(targetPx);
      return;
    }

    cancelAnimationFrame(rafId);

    const startPx = currentPx;
    const delta = targetPx - startPx;
    const startTime = performance.now();

    const easeOutCubic = (t) => 1 - Math.pow(1 - t, 3);

    const frame = (now) => {
      const elapsed = now - startTime;
      const progress = clamp(elapsed / duration, 0, 1);
      const eased = easeOutCubic(progress);
      render(startPx + delta * eased);

      if (progress < 1) {
        rafId = requestAnimationFrame(frame);
      }
    };

    rafId = requestAnimationFrame(frame);
  };

  const moveTo = (px) => animateTo(px);

  const moveBy = (deltaPx) => animateTo(currentPx + deltaPx);

  const moveLeft = (deltaPx = 20) => moveBy(-Math.abs(deltaPx));

  const moveRight = (deltaPx = 20) => moveBy(Math.abs(deltaPx));

  const moveToPercent = (percent) => {
    const px = clamp(percent, 0, 1) * getMaxPx();
    animateTo(px);
  };

  const moveToValue = (value) => {
    animateTo(valueToPx(clamp(value, min, max)));
  };

  const getValue = () => pxToValue(currentPx);

  const getState = () => ({
    px: currentPx,
    value: getValue(),
    percent: getMaxPx() === 0 ? 0 : currentPx / getMaxPx()
  });

  const pointerToPx = (clientX) => {
    const rect = track.getBoundingClientRect();
    const px = clientX - rect.left - handle.offsetWidth / 2;
    return clamp(px, 0, getMaxPx());
  };

  const onPointerDown = (e) => {
    dragging = true;
    handle.style.cursor = "grabbing";
    handle.setPointerCapture?.(e.pointerId);
    onStart(getState());
  };

  const onPointerMove = (e) => {
    if (!dragging) return;
    render(pointerToPx(e.clientX));
  };

  const onPointerUp = (e) => {
    if (!dragging) return;
    dragging = false;
    handle.style.cursor = "grab";
    handle.releasePointerCapture?.(e.pointerId);
    onEnd(getState());
  };

  const onTrackClick = (e) => {
    if (e.target === handle) return;
    moveTo(pointerToPx(e.clientX));
  };

  const onResize = () => {
    render(clamp(currentPx, 0, getMaxPx()));
  };

  handle.addEventListener("pointerdown", onPointerDown);
  window.addEventListener("pointermove", onPointerMove);
  window.addEventListener("pointerup", onPointerUp);
  track.addEventListener("click", onTrackClick);
  window.addEventListener("resize", onResize);

  render(valueToPx(initialValue));

  return {
    moveLeft,
    moveRight,
    moveBy,
    moveTo,
    moveToPercent,
    moveToValue,
    getValue,
    getState,
    setSmooth(enabled) {
      smooth = !!enabled;
    },
    setDuration(ms) {
      duration = Math.max(0, ms);
    },
    destroy() {
      cancelAnimationFrame(rafId);
      handle.removeEventListener("pointerdown", onPointerDown);
      window.removeEventListener("pointermove", onPointerMove);
      window.removeEventListener("pointerup", onPointerUp);
      track.removeEventListener("click", onTrackClick);
      window.removeEventListener("resize", onResize);
    }
  };
}

            humanLike: options.humanLike !== false,
            speed: options.speed || 'normal', // 'slow', 'normal', 'fast'
            debug: options.debug !== false,
            tolerance: options.tolerance || 5, // نطاق الخطأ المسموح (بكسل)
            ...options
        };

        this.isDragging = false;
        this.isCompleted = false;
        this.targetPosition = 0;
        this.currentPosition = 0;

        this._initElements();
        this._initEvents();
        this._calculateTarget();
        this._log('🚀 تم تهيئة محاكي السلايدر');
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // تهيئة العناصر
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _initElements() {
        this.els = {
            sliderArea: document.getElementById('sliderArea'),
            sliderBlock: document.getElementById('sliderBlock'),
            sliderTrack: document.getElementById('sliderTrack'),
            sliderText: document.getElementById('sliderText'),
            puzzlePiece: document.getElementById('puzzlePiece'),
            puzzleHole: document.getElementById('puzzleHole'),
            statusMsg: document.getElementById('statusMsg'),
            infoPanel: document.getElementById('infoPanel'),
            btnSimulate: document.getElementById('btnSimulate'),
            btnReset: document.getElementById('btnReset')
        };
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // أحداث المستخدم (يدعم الماوس واللمس)
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _initEvents() {
        const block = this.els.sliderBlock;

        // Mouse events
        block.addEventListener('mousedown', (e) => this._startDrag(e.clientX));
        document.addEventListener('mousemove', (e) => this._onDrag(e.clientX));
        document.addEventListener('mouseup', () => this._endDrag());

        // Touch events
        block.addEventListener('touchstart', (e) => this._startDrag(e.touches[0].clientX));
        document.addEventListener('touchmove', (e) => this._onDrag(e.touches[0].clientX));
        document.addEventListener('touchend', () => this._endDrag());
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // حساب موقع الهدف
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _calculateTarget() {
        const areaWidth = this.els.sliderArea.offsetWidth;
        const blockWidth = this.els.sliderBlock.offsetWidth;
        const maxPos = areaWidth - blockWidth;

        // الهدف: موقع اللغز (puzzle hole)
        const holeLeft = this.els.puzzleHole.offsetLeft;
        const pieceLeft = this.els.puzzlePiece.offsetLeft;

        // نحسب المسافة المطلوبة لتحريك القطعة لتتطابق مع الفتحة
        this.targetPosition = Math.max(0, Math.min(maxPos, holeLeft - pieceLeft + 50));
        this.maxPosition = maxPos;

        this._log(`🎯 الهدف المحسوب: ${this.targetPosition.toFixed(1)}px / ${maxPos.toFixed(1)}px`);
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // بدء السحب
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _startDrag(clientX) {
        if (this.isCompleted) return;
        this.isDragging = true;
        this.startX = clientX;
        this.startPos = this.currentPosition;
        this.els.sliderBlock.style.transition = 'none';
        this.els.sliderTrack.style.transition = 'none';
        this._log('👇 بدء السحب');
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // أثناء السحب
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _onDrag(clientX) {
        if (!this.isDragging) return;

        const delta = clientX - this.startX;
        let newPos = this.startPos + delta;
        newPos = Math.max(0, Math.min(this.maxPosition, newPos));

        this._updatePosition(newPos);
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // تحديث الموضع
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _updatePosition(pos) {
        this.currentPosition = pos;
        const percent = (pos / this.maxPosition) * 100;

        this.els.sliderBlock.style.left = `${pos}px`;
        this.els.sliderTrack.style.width = `${percent}%`;

        // تحريك قطعة اللغز مع السلايدر
        const pieceOffset = pos;
        this.els.puzzlePiece.style.transform = `translateY(-50%) translateX(${pieceOffset}px)`;

        // إخفاء النص عند البدء
        if (pos > 5) {
            this.els.sliderText.style.opacity = Math.max(0, 1 - (pos / 50));
        }
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // إنهاء السحب
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _endDrag() {
        if (!this.isDragging) return;
        this.isDragging = false;

        const diff = Math.abs(this.currentPosition - this.targetPosition);

        if (diff <= this.config.tolerance) {
            this._onSuccess();
        } else {
            this._onFail();
        }
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // نجاح
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _onSuccess() {
        this.isCompleted = true;
        this._updatePosition(this.targetPosition);

        this.els.sliderBlock.classList.add('success');
        this.els.sliderBlock.innerHTML = '✓';
        this.els.sliderText.textContent = '✅ تم التحقق بنجاح!';
        this.els.sliderText.style.opacity = '1';
        this.els.sliderText.style.color = '#00ff88';

        this._showStatus('🎉 تم التحقق بنجاح! CAPTCHA مكتمل.', 'success');
        this._log('✅ نجاح! التحقق مكتمل');
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // فشل
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _onFail() {
        this.els.sliderBlock.classList.add('fail');
        this._showStatus('❌ لم يتم التحقق. جرب مرة أخرى.', 'fail');
        this._log(`❌ فشل! المسافة: ${Math.abs(this.currentPosition - this.targetPosition).toFixed(1)}px`);

        setTimeout(() => {
            this.els.sliderBlock.classList.remove('fail');
            this._animateReset();
        }, 500);
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // إعادة تعيين متحركة
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _animateReset() {
        this.els.sliderBlock.style.transition = 'left 0.4s cubic-bezier(0.4, 0, 0.2, 1)';
        this.els.sliderTrack.style.transition = 'width 0.4s cubic-bezier(0.4, 0, 0.2, 1)';
        this._updatePosition(0);

        setTimeout(() => {
            this.els.sliderBlock.style.transition = '';
            this.els.sliderTrack.style.transition = '';
            this.els.sliderText.style.opacity = '1';
            this.els.sliderText.style.color = '';
        }, 400);
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // ═══════ المحاكاة التلقائية (القلب النابض) ═══════
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    async simulate(options = {}) {
        if (this.isCompleted || this.isDragging) return;

        const config = {
            humanLike: options.humanLike !== false,
            speed: options.speed || this.config.speed,
            randomOffset: options.randomOffset !== false,
            ...options
        };

        this.els.btnSimulate.disabled = true;
        this._log('🤖 بدء المحاكاة التلقائية...');

        // 1. حساب المسار
        let target = this.targetPosition;

        // إضافة عشوائية (محاكاة بشرية)
        if (config.randomOffset) {
            const offset = (Math.random() - 0.5) * this.config.tolerance * 2;
            target += offset;
            this._log(`🎲 إزاحة عشوائية: ${offset.toFixed(2)}px`);
        }

        // 2. تحديد السرعة
        const speedMap = { slow: 30, normal: 15, fast: 5 };
        const stepDelay = speedMap[config.speed] || 15;

        // 3. محاكاة الحركة البشرية (منحنى Bezier)
        const steps = 60;
        const startPos = 0;
        const endPos = Math.max(0, Math.min(this.maxPosition, target));

        this._log(`🎯 الهدف: ${endPos.toFixed(1)}px | الخطوات: ${steps} | السرعة: ${config.speed}`);

        // 4. تنفيذ الحركة
        for (let i = 0; i <= steps; i++) {
            const t = i / steps;
            // منحنى ease-out-cubic لمحاكاة الحركة البشرية
            const ease = 1 - Math.pow(1 - t, 3);
            const pos = startPos + (endPos - startPos) * ease;

            // إضافة اهتزاز طفيف (tremor)
            let finalPos = pos;
            if (config.humanLike && i > steps * 0.3 && i < steps * 0.9) {
                finalPos += (Math.random() - 0.5) * 2;
            }

            this._updatePosition(finalPos);

            // تأخير متغير
            const delay = stepDelay + (config.humanLike ? Math.random() * 10 : 0);
            await this._sleep(delay);
        }

        // 5. محاكاة الأحداث
        this._simulateEvents(endPos);

        // 6. التحقق
        const diff = Math.abs(this.currentPosition - this.targetPosition);
        if (diff <= this.config.tolerance) {
            this._onSuccess();
        } else {
            this._onFail();
        }

        this.els.btnSimulate.disabled = false;
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // محاكاة أحداث DOM الحقيقية
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _simulateEvents(finalPos) {
        const block = this.els.sliderBlock;
        const rect = block.getBoundingClientRect();
        const centerX = rect.left + rect.width / 2;
        const centerY = rect.top + rect.height / 2;

        const eventInit = {
            bubbles: true,
            cancelable: true,
            view: window,
            clientX: centerX,
            clientY: centerY
        };

        // mousedown
        block.dispatchEvent(new MouseEvent('mousedown', eventInit));
        this._log('📤 dispatch: mousedown');

        // mousemove (محاكاة)
        for (let i = 0; i < 5; i++) {
            block.dispatchEvent(new MouseEvent('mousemove', {
                ...eventInit,
                clientX: centerX + (Math.random() - 0.5) * 3
            }));
        }

        // mouseup
        block.dispatchEvent(new MouseEvent('mouseup', eventInit));
        this._log('📤 dispatch: mouseup');

        // click
        block.dispatchEvent(new MouseEvent('click', eventInit));
        this._log('📤 dispatch: click');
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // إعادة تعيين
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    reset() {
        this.isCompleted = false;
        this.isDragging = false;
        this.currentPosition = 0;

        this.els.sliderBlock.classList.remove('success', 'fail');
        this.els.sliderBlock.innerHTML = '➜';
        this.els.sliderBlock.style.left = '0px';
        this.els.sliderBlock.style.transition = '';

        this.els.sliderTrack.style.width = '0%';
        this.els.sliderTrack.style.transition = '';

        this.els.sliderText.textContent = 'اسحب لليمين لإكمال اللغز ←';
        this.els.sliderText.style.opacity = '1';
        this.els.sliderText.style.color = '';

        this.els.puzzlePiece.style.transform = 'translateY(-50%) translateX(0px)';

        this.els.statusMsg.classList.remove('show');
        this.els.btnSimulate.disabled = false;

        // إعادة حساب الهدف (للتنوع)
        this._randomizePuzzle();
        this._calculateTarget();

        this._log('🔄 تم إعادة التعيين');
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // تنويع موقع اللغز
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _randomizePuzzle() {
        const container = this.els.puzzleImage;
        const maxLeft = container.offsetWidth - 100;
        const randomLeft = 50 + Math.random() * (maxLeft - 100);
        this.els.puzzleHole.style.left = `${randomLeft}px`;
        this.els.puzzlePiece.style.left = `${20 + Math.random() * 30}px`;
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // عرض رسالة الحالة
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _showStatus(text, type) {
        const msg = this.els.statusMsg;
        msg.textContent = text;
        msg.className = `status-msg ${type} show`;
    }

    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    // تسجيل السجل
    // ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
    _log(message, type = 'sim') {
        if (!this.config.debug) return;

        const time = new Date().toLocaleTimeString('ar-SA', { hour12: false });
        const panel = this.els.infoPanel;

        const colors = {
            sim: 'log-sim',
            info: 'log-info',
            success: 'log-success',
            error: 'log-error'
        };

        const line = document.createElement('div');
        line.innerHTML = `<span class="log-time">[${time}]</span> <span class="${colors[type] || colors.sim}">${message}</span>`;
        panel.appendChild(line);
        panel.scrollTop = panel.scrollHeight;

        console.log(`[SliderSim] ${message}`);
    }
