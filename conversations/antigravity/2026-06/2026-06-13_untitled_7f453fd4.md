import subprocess
import sys
import time

def run_queue():
    while True:
        with open("prompts_queue.txt", "r", encoding="utf-8") as f:

        if not prompts:
            print("[*] الطابور فاضي.. انتظار 10 ثواني...")
            time.sleep(10)
            continue

        current_prompt = prompts[0]
        print(f"🎬 جاري توليد: {current_prompt}")

        # تشغيل سكريبت التوليد
        proc = subprocess.run([sys.executable, "03.07_generate_video_grok.py", "--prompt", current_prompt])

        # بعد ما يخلص (سواء نجح أو فشل)، نشيل البرومبت من الطابور
        with open("prompts_queue.txt", "r", encoding="utf-8") as f:
            remaining = f.readlines()

            f.writelines(remaining[1:])
