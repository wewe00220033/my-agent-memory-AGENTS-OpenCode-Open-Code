class PureFollowersEngine:
    def __init__(self):
        # استخدام جلسة موحدة للحفاظ على الكوكيز والبصمة
        self.session = requests.Session(impersonate="chrome120")

    def fetch_from_social_tracker(self, username):
        """المصدر الأول: Social-Tracker عبر Next.js Server Action المباشر"""
        try:
            url = f"https://social-tracker.com/stats/instagram/follower/{username}"
            headers = {
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
                "Next-Action": "40d509ecd64f8f4a8ec70fd10196d1e36d6df3c78a",
                "Content-Type": "text/plain;charset=UTF-8",
                "Origin": "https://social-tracker.com",
                "Referer": url
            }
            payload = json.dumps([username])
            res = self.session.post(url, data=payload, headers=headers, timeout=10)

            if res.status_code == 200:
                match = re.search(r'\{"ok":true[^\n]+', res.text)
                if match:
                    data = json.loads(match.group(0)).get("data", {})
                    return {
                        "provider": "Social-Tracker",
                        "username": username,
                        "full_name": data.get("fullName", username),
                        "followers": data.get("followersCount", 0),
                        "following": data.get("followsCount", 0),
                        "posts": data.get("postsCount", 0),
                        "is_private": data.get("private", False)
                    }
        except Exception:
            pass
        return None

    def fetch_from_pathsocial(self, username):
        """المصدر الثاني: PathSocial المباشر"""
        try:
            url = "https://www.pathsocial.com/wp-json/pathsocial/v1/profile-analyzer"
            headers = {
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",

                "Origin": "https://www.pathsocial.com",
                "Referer": "https://www.pathsocial.com/free-instagram-tools/instagram-profile-analyzer/"
            }
            payload = {"username": username}
            res = self.session.post(url, json=payload, headers=headers, timeout=10)

            if res.status_code == 200:
                data = res.json()
                if data.get("success") and "data" in data:
                    pdata = data["data"]
                    return {
                        "provider": "PathSocial",
                        "username": username,
                        "full_name": pdata.get("full_name", username),
                        "followers": pdata.get("followers", 0),
                        "following": pdata.get("following", 0),
                        "posts": pdata.get("posts", 0),
                        "is_private": pdata.get("is_private", False)
                    }
        except Exception:
            pass
        return None

    def get_followers(self, username):
        username = clean_username(username)
        print(f"\n{COLOR_CYAN}🔍 فحص الحساب: @{username} ...{COLOR_RESET}")

        # قائمة المزودين بالترتيب
        providers = [
            self.fetch_from_social_tracker,
            self.fetch_from_pathsocial
        ]

        for provider_func in providers:
            result = provider_func(username)
            if result:
                print(f"{COLOR_GREEN}✅ تم الجلب بنجاح عبر [{result['provider']}]!{COLOR_RESET}")
                print(f"👤 الاسم: {COLOR_BOLD}{result['full_name']}{COLOR_RESET}")
                print(f"📊 المتابعين: {COLOR_YELLOW}{result['followers']:,}{COLOR_RESET}")
                print(f"➡️ المتابَعين: {result['following']:,}")
                print(f"📸 البوستات: {result['posts']:,}")
                print(f"🔒 خاص: {'نعم 🔒' if result['is_private'] else 'لا 🌐'}")
                return result

        print(f"{COLOR_RED}❌ تعذر جلب البيانات من كافة المصادر المتاحة.{COLOR_RESET}")
        return None

if __name__ == "__main__":
    engine = PureFollowersEngine()
    if len(sys.argv) > 1:

            engine.get_followers(u)

        if user_input:
