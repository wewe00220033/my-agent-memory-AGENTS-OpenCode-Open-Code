### 🛠️ كيفية تحقق وتنسيق الأرقام برمجياً (Python Example)
لضمان إرسال أرقام الهواتف بالتنسيق الصحيح وتجنب الأخطاء البرمجية أثناء التسجيل، يمكن استخدام مكتبات معالجة الهواتف القياسية مثل `phonenumbers` (وهي البورت المعتمد لمكتبة Google libphonenumber الشهيرة):

```python
import phonenumbers
from phonenumbers import PhoneNumberFormat

def validate_and_format_phone(raw_number, default_country="HT"):
    """
    وظيفة للتحقق من صحة الرقم وتنسيقه دولياً.
    default_country: كود الدولة الافتراضي بصيغة ISO (مثل HT لهايتي أو EG لمصر)
    """
    try:
        # 1. تحليل الرقم الخام مع تحديد الدولة الافتراضية
        parsed_number = phonenumbers.parse(raw_number, default_country)

        # 2. التحقق من أن الرقم حقيقي وصحيح في الدولة المحددة
        if not phonenumbers.is_valid_number(parsed_number):
            return {
                "success": False,
                "error": "الرقم غير صالح للدولة المحددة"
            }

        # 3. تنسيق الرقم بالصيغة الدولية E.164 (تبدأ بـ + وكود الدولة)
        formatted_number = phonenumbers.format_number(parsed_number, PhoneNumberFormat.E164)

        return {
            "success": True,
            "formatted": formatted_number,
            "country_code": parsed_number.country_code,
            "national_number": parsed_number.national_number
        }

    except phonenumbers.NumberParseException as e:
        return {
            "success": False,
            "error": f"فشل في تحليل الرقم: {str(e)}"
        }

# مثال للاستخدام:
# رقم خام مكتوب بدون كود الدولة مع تحديد هايتي كدولة افتراضية

    print(f"كود الدولة: {result['country_code']}")
else:
    print(f"خطأ: {result['error']}")
```

### 🛡️ آليات الدفاع ضد الأتمتة المباشرة في الـ Backend

الأنظمة الكبيرة بتطبق عدة طبقات حماية للتأكد من إن الطلب جاي من مستخدم حقيقي عبر المتصفح مش من سكربت خارجي:

1. **توكنات الـ CSRF (مثل `fb_dtsg` و `lsd`):**

2. **تقييد المعدل والـ IP Rate Limiting:**
   * **فكرتها:** تتبع عدد الطلبات الموجهة للـ API الحساس (زي طلب الـ SMS) من نفس الـ IP أو الحساب. لو زاد عن حد معين في الدقيقة (مثلاً طلبين كل 5 دقائق)، السيرفر بيرفض الطلب مباشرة بكود `429 Too Many Requests` أو يوجه المستخدم لتحدي كابتشا.

3. **فحص بصمة العميل (Client Fingerprinting & JA3):**
   * **فكرتها:** السيرفر بيفحص طريقة مصافحة الـ TLS والـ HTTP/2 الخاصة بالمتصفح. السكربتات العادية (زي مكتبة Python requests) بيكون ليها بصمة مختلفة تماماً عن المتصفحات الحقيقية (مثل Chrome)، والسيرفر بيحظرها فوراً قبل معالجة الطلب.

### 💻 نموذج برمجي لحماية الـ API ضد البوتات (Python API Guard)

إليك مثال لكيفية قيام المطورين ببناء نقطة نهاية (Endpoint) آمنة تتحقق من الجلسة وتمنع الأتمتة السريعة باستخدام إطار العمل `FastAPI`:

```python
from fastapi import FastAPI, Header, HTTPException, Depends
from cachetools import TTLCache

app = FastAPI()

# كاش مؤقت في الذاكرة لتتبع الطلبات لكل IP لمنع الإرسال المكثف (Rate Limiter)
# صلاحيته دقيقة واحدة
rate_limit_cache = TTLCache(maxsize=10000, ttl=60)

# قاعدة بيانات وهمية للجلسات النشطة وتوكنات الـ CSRF المطابقة لها
active_sessions = {
    "session_token_user_1": {
        "user_id": "61591450487577",
        "csrf_token": "[REDACTED:KV_SECRET]"
    }
}

def verify_session_and_csrf(
    cookie_session: str = Header(None, alias="Cookie"),
    x_csrf_token: str = Header(None, alias="X-CSRF-Token"),
    client_ip: str = Header("127.0.0.1", alias="X-Forwarded-For")
):
    # 1. فحص حد الطلبات (Rate Limit Check)
    request_count = rate_limit_cache.get(client_ip, 0)
    if request_count >= 3:  # حد أقصى 3 طلبات في الدقيقة
        raise HTTPException(status_code=429, detail="لقد تجاوزت الحد المسموح به من الطلبات")
    rate_limit_cache[client_ip] = request_count + 1

    # 2. استخراج وفحص كوكيز الجلسة
    session_id = None
    if cookie_session:
        for item in cookie_session.split(";"):
            if "session_id=" in item:
                session_id = item.split("=")[1].strip()

    if not session_id or session_id not in active_sessions:
        raise HTTPException(status_code=401, detail="جلسة غير صالحة أو منتهية")

    # 3. التحقق من توكن الـ CSRF المرفق في الـ Headers
    expected_csrf = active_sessions[session_id]["csrf_token"]
    if not x_csrf_token or x_csrf_token != expected_csrf:
        raise HTTPException(status_code=403, detail="فشل التحقق الأمني (CSRF Failed)")

    return active_sessions[session_id]["user_id"]
```
