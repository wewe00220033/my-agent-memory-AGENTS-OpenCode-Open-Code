def send_telegram_message(text):
    """
    إرسال إشعار تليجرام بتنسيق HTML باستخدام urllib المدمجة فقط.
    - خفيفة جداً على الرامات: بدون requests أو أي مكتبة خارجية.
    - أي فشل في الإرسال لا يوقف السكريبت (إشعار اختياري).
    """
    if not TELEGRAM_BOT_TOKEN or not TELEGRAM_CHAT_ID:
        return False
    api_url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    payload = urllib.parse.urlencode({
        "chat_id": TELEGRAM_CHAT_ID,
        "text": text,
        "parse_mode": "HTML",
        "disable_web_page_preview": "true",
    }).encode("utf-8")
    try:
        req = urllib.request.Request(
            api_url, data=payload,
            headers={"Content-Type": "application/x-www-form-urlencoded"}
        )
        with urllib.request.urlopen(req, timeout=15) as resp:
            if resp.status == 200:
                log_message("تم إرسال إشعار تليجرام بنجاح. 📨", GREEN)
                return True
            log_message(f"⚠️ تليجرام أعاد حالة غير متوقعة: {resp.status}", YELLOW)
            return False
    except Exception as e:
        log_message(f"⚠️ تعذر إرسال إشعار تليجرام: {e}", YELLOW)
        return False

إليك التغيير **قبل وبعد** في السكريبت عشان يتخطى مجلد `.github`:

### 📍 التعديل في دالة نسخ شجرة الملفات (`copy_tree`):

**قبل:**
```python
        if item == ".git":
            continue
```

**بعد (تخطي مجلدات .git و .github معاً):**
```python
        if item in (".git", ".github"):
            continue
```

### 🛡️ إزاي التعديل ده بيحل المشكلة؟
مجلد الـ workflows اللي جواه ملف الـ `ci.yml` بيكون موجود بالأساس جوه مجلد `.github`.
دلوقتي لما السكريبت بيفك الملف المضغوط وينسخه للمستودع، هيتجاهل مجلد `.github` تماماً ولن يقوم بنسخه أو تعديله، وبالتالي لما يجي يعمل `git push` مش هيلمس الـ workflows خالص، وجيت هاب هيقبل الرفع فوراً بدون أي مشكلة في الصلاحيات.
