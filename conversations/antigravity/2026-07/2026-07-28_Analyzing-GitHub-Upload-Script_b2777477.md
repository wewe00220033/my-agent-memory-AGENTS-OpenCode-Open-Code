```python
from rich.progress import Progress, SpinnerColumn, TextColumn, BarColumn, TimeRemainingColumn
import time

with Progress(
    SpinnerColumn("dots"),                           # أيقونة لودينج متحركة
    TextColumn("[bold cyan]{task.description}"),     # النص الملون
    BarColumn(bar_width=30),                         # شريط التقدم
    TextColumn("[bold yellow]{task.percentage:>3.0f}%"), # النسبة المئوية
    TimeRemainingColumn(),                           # الوقت المتبقي

    while not progress.finished:
        time.sleep(1)
        progress.update(task, advance=1)
```
