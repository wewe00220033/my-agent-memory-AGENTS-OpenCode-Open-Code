    """طلب التوكن من سيرفر EzSolver المحلي"""
    payload = {
        "siteurl": site_url,      # خلي بالك: اسمها siteurl مش url
        "sitekey": sitekey,
        "timeout": timeout
    }

    try:
        # بنبعت طلب POST للسيرفر المحلي

        response.raise_for_status()
        data = response.json()

        if "token" in data:
            print("✅ تم استلام التوكن بنجاح من السيرفر المحلي!")
            return data["token"]
        else:
            raise ValueError(f"السيرفر رجّع خطأ: {data.get('error')}")
