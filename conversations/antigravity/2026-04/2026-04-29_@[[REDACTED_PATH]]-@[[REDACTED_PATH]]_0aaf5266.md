```python
import json
import os

def filter_accounts(input_file, output_file):
    # قراءة الملف الأصلي
    with open(input_file, 'r', encoding='utf-8') as f:
        data = json.load(f)

    # فلترة الحسابات لاستخراج الإيميل والباسورد فقط
    filtered_accounts = [
        {"email": acc["email"], "password": acc["password"]}
        for acc in data if "email" in acc and "password" in acc
    ]

    # التنسيق النهائي المطلوب
    result = {"accounts": filtered_accounts}

    # حفظ الملف الجديد
    with open(output_file, 'w', encoding='utf-8') as f:
        json.dump(result, f, indent=4)

if __name__ == "__main__":
    filter_accounts("accounts_genspark.json", "accounts_filtered.json")
```
